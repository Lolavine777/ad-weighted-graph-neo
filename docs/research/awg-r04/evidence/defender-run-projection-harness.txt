"""Hash complete native attempts per goal without loading both large runs at once."""
import argparse
import hashlib
import json
import subprocess
from pathlib import Path

p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--folder', type=Path, required=True)
p.add_argument('--repo', type=Path, required=True)
p.add_argument('--output', type=Path, required=True)
a = p.parse_args()
assert not a.output.exists()
sha = lambda b: 'sha256:' + hashlib.sha256(b).hexdigest()
canon = lambda value: sha(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode())
def pin(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        while chunk := f.read(1048576):
            h.update(chunk)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': 'sha256:' + h.hexdigest()}

folder = a.folder.resolve()
with (folder / 'result.json').open() as f:
    result = json.load(f)
request = json.loads((folder / 'request.json').read_text())
summary = json.loads((folder / 'summary.json').read_text())
inventory = json.loads((folder / 'inventory.json').read_text())
native_inputs = result['reproducibility']['inputs']
assert canon(native_inputs['request']) == native_inputs['request_digest']
revision = result['reproducibility']['revision']
# Independently check every recorded executable/config pin against its Git object.
git = subprocess.Popen(['git', 'cat-file', '--batch'], cwd=a.repo, stdin=subprocess.PIPE, stdout=subprocess.PIPE)
for name, digest in native_inputs['source_pins'].items():
    git.stdin.write((revision + ':' + name + '\n').encode()); git.stdin.flush()
    header = git.stdout.readline().decode().split()
    assert header[-1] != 'missing', name
    raw = git.stdout.read(int(header[-1])); assert git.stdout.read(1) == b'\n'
    assert sha(raw) == digest, name
git.stdin.close(); assert git.wait() == 0
assert summary['batch']['declared_cells'] == 12 and len(request['sources']) == 1
source = request['sources'][0]['node_id']
assert len(inventory['targets']) == len(request['target_manifest']['targets']) == 12
assert {t['goal']['goal_id'] for t in inventory['targets']} == {t['goal_id'] for t in request['target_manifest']['targets']}
targets = {t['goal_id']: t for t in request['target_manifest']['targets']}
rows = []
for sid, matrix in result['source_goal_matrix'].items():
    for gid, cell in matrix.items():
        key = canon([sid, gid]); trace = native_inputs['public_queries'][key]
        query, native = trace['query'], trace['result']
        state = trace['materialized']['initial_attack_state']
        assert state['usable_identity'] == [[source, source]]
        assert not any(state[k] for k in ['control', 'admin_control', 'recovered_secret', 'reset_password_control'])
        semantic_query = dict(query)
        scenario = semantic_query.pop('scenario_reference')
        assert scenario.startswith('batch-cell:sha256:')
        resolution = semantic_query.pop('rating_resolution', None)
        profile_status = semantic_query.pop('effort_profile_metadata', None)
        paths = []
        for path in native['paths']:
            path = dict(path)
            assert path.pop('scenario_reference') == scenario
            paths.append(path)
        ledger = []
        if cell['min_effort'] is not None:
            cid = cell['min_effort_path']['candidate_id']
            units = native['rating_coverage']['candidates'][cid]['effort']['units']
            for uid, unit in units.items():
                k = unit['rating_audit']['lookup_key']
                assert k['scope'] == scenario and k['auth_context'] == source
                assert k['policy_id'] == 'reset-activation-c-v2'
                assert k['technique'] in {'scenario_reset_password', 'acquire_reset_identity'}
                assert unit['cost'] == (1 if k['technique'] == 'scenario_reset_password' else 0)
                ledger.append({'technique': k['technique'], 'actor': k['actor_id'], 'source': k['source_id'],
                               'target': k['target_id'], 'context': k['auth_context'], 'cost': unit['cost'],
                               'unit_id': uid, 'numeric_basis_class': unit['rating_audit']['numeric_basis_class']})
            # Independent one-reset oracle: concrete owner/context, no native unit-ID helper.
            owner = 'S-1-5-21-1271839998-3495686680-2211690266-2799'
            expected = [('scenario_reset_password', source, source, owner, source, 1),
                        ('acquire_reset_identity', owner, owner, source, source, 0)]
            assert sorted((u['technique'], u['actor'], u['source'], u['target'], u['context'], u['cost']) for u in ledger) == sorted(expected)
            assert sum(u['cost'] for u in ledger) == cell['min_effort'] == 1
        rows.append({'source_id': sid, 'goal_id': gid, 'target': targets[gid],
                     'status': cell['status'], 'reached': cell['final_goal_reached'], 'C': cell['min_effort'],
                     'P': cell['likelihood_interval'], 'impact': cell['impact'], 'expansions': cell['expansions'],
                     'guarantees': trace['guarantees'], 'query_without_verified_rating_scope_sha256': canon(semantic_query),
                     'actual_scenario': scenario, 'rating_resolution_sha256': canon(resolution),
                     'effort_applicability': profile_status, 'complete_attempts_sha256': canon(native['attempts']),
                     'paths_without_verified_scenario_sha256': canon(paths),
                     'witness_requests': [[s['result']['request'] for s in path['transition_history']] for path in paths],
                     'ledger': ledger})
out = {'version': 'awg-r04-defender-run-projection/1', 'revision': revision,
       'all_source_pins_verified_against_git_objects': True, 'source_pins': native_inputs['source_pins'],
       'summary': summary, 'request': request,
       'inventory_without_implementation_pin_sha256': canon({k: v for k, v in inventory.items() if k != 'implementation_sha256'}),
       'rows': rows, 'artifacts': {n: pin(folder / n) for n in ['result.json', 'checkpoint.json', 'summary.json', 'overview.md']},
       'normalization': 'Only rating_resolution, effort_profile_metadata and the verified per-cell scenario reference are removed from cross-profile query/path comparison. Attempts are hashed whole, unchanged. Different C profiles intentionally have different scenario/work identities.',
       'limits': 'One fixed strong-ACL Computer, seven declared runtime assumptions,12 automatic important goals; not domain-wide/all-technique coverage, calibrated P, speedup or optimum proof.'}
a.output.write_text(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'revision': revision, 'goals': len(rows), 'rated': sum(r['C'] is not None for r in rows)}))
