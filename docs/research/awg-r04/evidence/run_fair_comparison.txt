def union_cost(ids):
    keys={w for action_id in ids for w in actions[action_id]['referenced_work_unit_ids']}
    return sum(units[k]['computed_cost'] for k in keys)
weights={a['edge_id']:union_cost([a['action_id']]) for a in rating['bound_actions']}

bindings={(k[0],*(query['aliases'].get(x,x) for x in k[1:])):v for k,v in query['action_bindings']}
def projection(weighted,condition=True):
    edges=[e for e in inputs['graph']['edges'] if condition or e['kind']!='AdminTo']
    adjacency={}
    for e in edges:adjacency.setdefault(e['source'],[]).append(e)
    if weighted:
        pending=[(0,0,source,[],[])];best={source:0};serial=1
        while pending:
            cost,_,node,nodes,ids=heapq.heappop(pending)
            if cost!=best[node]:continue
            if node==target:return {'nodes':nodes+[node],'edge_ids':ids,'hops':len(ids),'C':cost}
            for e in adjacency.get(node,[]):
                value=cost+weights[e['id']]
                if value<best.get(e['target'],float('inf')):
                    best[e['target']]=value;heapq.heappush(pending,(value,serial,e['target'],nodes+[node],ids+[e['id']]));serial+=1
    else:
        pending=deque([(source,[],[])]);seen={source}
        while pending:
            node,nodes,ids=pending.popleft()
            if node==target:return {'nodes':nodes+[node],'edge_ids':ids,'hops':len(ids),'C':sum(weights[e] for e in ids)}
            for e in adjacency.get(node,[]):
                if e['target'] not in seen:seen.add(e['target']);pending.append((e['target'],nodes+[node],ids+[e['id']]))
    return None

bfs=projection(False);dijkstra=projection(True)
assert bfs['hops']==1 and bfs['C']==5 and dijkstra['hops']==2 and dijkstra['C']==2
assert projection(False,False) is projection(True,False) is None
