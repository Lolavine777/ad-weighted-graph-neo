def query_from_capture(path, key):
    import re
    with path.open(encoding='utf-8') as f:
        marker=json.dumps(key)
        buf=''
        while True:
            chunk=f.read(1048576);assert chunk,'query key missing'
            buf+=chunk
            pos=buf.find(marker)
            if pos>=0:
                buf=buf[pos+len(marker):];break
            buf=buf[-len(marker):]
        while True:
            match=re.search(r'"query"\s*:',buf)
            if match:
                buf=buf[match.end():].lstrip();break
            chunk=f.read(1048576);assert chunk,'query missing';buf+=chunk
        while True:
            try:
                query,_=json.JSONDecoder().raw_decode(buf);return query
            except json.JSONDecodeError:
                chunk=f.read(1048576);assert chunk,'truncated query';buf+=chunk
