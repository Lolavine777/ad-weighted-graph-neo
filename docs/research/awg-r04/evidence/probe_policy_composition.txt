"""Finite expected identity chain, with disconnected padding across the policy threshold."""
import copy
import json
import sys
from dataclasses import replace
from pathlib import Path

repo=Path(__file__).resolve().parent/'repo'
sys.path[:0]=[str(repo),str(repo/'tests')]
from test_computer_foothold_boundary import CASES, TECHNIQUES
from test_machine_identity_profile import scenario,request
from test_engine import ACTOR, GROUP, HOST, _edge
from src.engine.graph import EngineGraph
from src.engine.batch import run_source_goal_batch
from src.engine.semantics.scenario_policy import SEAMLESS_STANDARD_PROFILE

rows=[]
for padding in [0,201]:
 graph,q=scenario(CASES[2]);raw=graph.to_canonical()
 for i in range(padding):
  template=copy.deepcopy(graph.node(ACTOR));template.update(id=f'S-1-5-21-1000-2000-3000-{8000+i}',label=f'Uncontrolled{i}')
  raw['nodes'].append(template);raw['edges'].append(_edge(template['id'],GROUP,'MemberOf'))
 graph=EngineGraph.from_canonical(raw);req=request(graph,q)
 for s in req['sources']:s['declared_assumptions']=[]
 policy=replace(SEAMLESS_STANDARD_PROFILE,bound_snapshot_id=graph.metadata['snapshot_id'])
 result=run_source_goal_batch(req,graph,allowed_techniques=TECHNIQUES,scenario_policy=policy,effort_profile='reset-activation-c-v2').to_dict()
 cell=next(iter(next(iter(result['source_goal_matrix'].values())).values()))
 trace=next(iter(result['reproducibility']['inputs']['public_queries'].values()))
 rows.append({'padding':padding,'edges':len(graph.edges()),'reached':cell['final_goal_reached'],'status':cell['status'],'C':cell['min_effort'],'assumptions':trace['query']['assumptions'],'unknown':sorted({p for a in trace['result']['attempts'] for p in a['unknown']}),'expansions':cell['expansions']})
Path(sys.argv[1]).write_text(json.dumps(rows,indent=2)+'\n')
print(json.dumps([{k:v for k,v in r.items() if k!='assumptions'} for r in rows]))
